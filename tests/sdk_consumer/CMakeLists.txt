cmake_minimum_required(VERSION 3.25)
project(SyrnckInstalledConsumer LANGUAGES C)
if(NOT SYRNCK_CONSUMER_C_ONLY)
  enable_language(CXX)
endif()
enable_testing()
find_package(Syrnck 0.1 CONFIG REQUIRED COMPONENTS Core Git C)
if(NOT SYRNCK_CONSUMER_C_ONLY)
  add_executable(cpp_consumer core.cpp)
  target_link_libraries(cpp_consumer PRIVATE Syrnck::core Syrnck::git)
  add_test(NAME cpp_consumer COMMAND cpp_consumer "${CMAKE_CURRENT_BINARY_DIR}/cpp-store")
  set(abi_targets ffi ffi_static)
else()
  set(abi_targets ffi)
endif()
foreach(kind ${abi_targets})
  add_executable(${kind}_consumer abi.c)
  target_link_libraries(${kind}_consumer PRIVATE Syrnck::${kind})
  if(WIN32 AND kind STREQUAL "ffi")
    add_custom_command(TARGET ${kind}_consumer POST_BUILD COMMAND "${CMAKE_COMMAND}" -E
      copy_if_different "$<TARGET_FILE:Syrnck::ffi>" "$<TARGET_FILE_DIR:${kind}_consumer>")
  endif()
  add_test(NAME ${kind}_consumer COMMAND ${kind}_consumer "${CMAKE_CURRENT_BINARY_DIR}/${kind}-store")
endforeach()
if(SYRNCK_CONSUMER_NETWORK AND NOT SYRNCK_CONSUMER_C_ONLY)
  find_package(Syrnck CONFIG REQUIRED COMPONENTS Network)
  add_executable(net_consumer net.cpp)
  target_link_libraries(net_consumer PRIVATE Syrnck::git_net)
  add_test(NAME net_consumer COMMAND net_consumer)
endif()
