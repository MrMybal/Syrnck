cmake_minimum_required(VERSION 3.25)
project(Syrnck VERSION 0.1.0 LANGUAGES C CXX)
# CMake 3.31 can misdecode French MSVC's UTF-8 /showIncludes prefix as OEM.
# Ninja needs the actual prefix to record header dependencies correctly.
if(MSVC AND CMAKE_CL_SHOWINCLUDES_PREFIX MATCHES "^Remarque")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "Remarque : inclusion du fichier : ")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(SYRNCK_BUILD_TESTS "Build regression tests" ON)
option(SYRNCK_BUILD_BENCHMARKS "Build the synthetic chunk benchmark" ON)
option(SYRNCK_TEST_GO "Test the Go binding when Go is available" ON)
option(SYRNCK_SANITIZE "Enable ASan/UBSan with GCC/Clang on Unix" OFF)
if(SYRNCK_SANITIZE AND MSVC)
  message(FATAL_ERROR "SYRNCK_SANITIZE currently requires GCC/Clang on Unix; it is not implemented for MSVC")
endif()
option(SYRNCK_BUILD_NETWORK "Build the HTTP server/client library" ON)
option(SYRNCK_BUILD_CLI "Build the command-line executable" ON)
if(SYRNCK_BUILD_CLI AND NOT SYRNCK_BUILD_NETWORK)
  message(FATAL_ERROR "The CLI requires SYRNCK_BUILD_NETWORK=ON; disable both for a core-only build")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/blake3 EXCLUDE_FROM_ALL)
add_library(monocypher STATIC third_party/monocypher/monocypher.c)
target_include_directories(monocypher PUBLIC third_party/monocypher)
add_library(utf8proc STATIC third_party/utf8proc/utf8proc.c)
target_compile_definitions(utf8proc PUBLIC UTF8PROC_STATIC)
target_include_directories(utf8proc PUBLIC third_party/utf8proc)
include(cmake/dependencies.cmake)
find_package(Threads REQUIRED)

add_library(syrnck_core STATIC src/model.cpp src/platform.cpp src/chunker.cpp src/store.cpp src/object_cache.cpp src/pack_storage.cpp src/pack_index.cpp src/revision.cpp src/i18n.cpp src/settings.cpp)
add_library(Syrnck::core ALIAS syrnck_core)
target_compile_features(syrnck_core PUBLIC cxx_std_20)
target_include_directories(syrnck_core PUBLIC include third_party PRIVATE src third_party/fastcdc)
target_link_libraries(syrnck_core PRIVATE BLAKE3::blake3 monocypher utf8proc libzstd_static Threads::Threads)
if(WIN32)
  target_link_libraries(syrnck_core PRIVATE bcrypt)
endif()
add_library(syrnck_git STATIC src/git.cpp)
target_link_libraries(syrnck_git PUBLIC syrnck_core)
target_include_directories(syrnck_git PRIVATE src)
if(SYRNCK_BUILD_NETWORK)
  add_library(syrnck_net STATIC src/net.cpp)
  target_link_libraries(syrnck_net PUBLIC syrnck_core PRIVATE CURL::libcurl Threads::Threads monocypher)
  target_include_directories(syrnck_net SYSTEM PRIVATE third_party/httplib)
  if(WIN32)
    target_link_libraries(syrnck_net PRIVATE ws2_32 crypt32)
  endif()
endif()
add_library(syrnck_ffi SHARED src/ffi.cpp)
target_include_directories(syrnck_ffi PUBLIC include)
target_compile_definitions(syrnck_ffi PRIVATE SYRNCK_EXPORTS)
target_link_libraries(syrnck_ffi PRIVATE syrnck_core)
add_library(syrnck_ffi_static STATIC src/ffi.cpp)
target_include_directories(syrnck_ffi_static PUBLIC include)
target_compile_definitions(syrnck_ffi_static PUBLIC SYRNCK_STATIC)
target_link_libraries(syrnck_ffi_static PRIVATE syrnck_core)
if(SYRNCK_BUILD_CLI)
  add_executable(syrnck src/main.cpp src/cli_control.cpp)
  target_link_libraries(syrnck PRIVATE syrnck_git syrnck_net)
endif()
set(SYRNCK_TARGETS syrnck_core syrnck_git syrnck_ffi syrnck_ffi_static)
if(SYRNCK_BUILD_NETWORK)
  list(APPEND SYRNCK_TARGETS syrnck_net)
endif()
if(SYRNCK_BUILD_CLI)
  list(APPEND SYRNCK_TARGETS syrnck)
endif()
foreach(target ${SYRNCK_TARGETS})
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /permissive- /EHsc)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(SYRNCK_SANITIZE)
      target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
include(GNUInstallDirs)
install(TARGETS ${SYRNCK_TARGETS}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES third_party/nlohmann/json.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/nlohmann)
if(SYRNCK_BUILD_BENCHMARKS)
  add_executable(syrnck_bench examples/chunk_bench.cpp)
  target_link_libraries(syrnck_bench PRIVATE syrnck_core)
  add_executable(syrnck_read_bench examples/read_bench.cpp)
  target_link_libraries(syrnck_read_bench PRIVATE syrnck_core)
  add_executable(syrnck_index_bench examples/index_bench.cpp)
  target_include_directories(syrnck_index_bench PRIVATE src tests)
  target_link_libraries(syrnck_index_bench PRIVATE syrnck_core BLAKE3::blake3)
  add_executable(syrnck_gc_bench examples/gc_bench.cpp)
  target_include_directories(syrnck_gc_bench PRIVATE src)
  target_link_libraries(syrnck_gc_bench PRIVATE syrnck_core BLAKE3::blake3 libzstd_static)
  add_executable(syrnck_compact_bench examples/compact_bench.cpp)
  target_link_libraries(syrnck_compact_bench PRIVATE syrnck_core)
  if(WIN32)
    target_link_libraries(syrnck_index_bench PRIVATE psapi)
    target_link_libraries(syrnck_gc_bench PRIVATE psapi)
    target_link_libraries(syrnck_compact_bench PRIVATE psapi)
  endif()
endif()

  enable_testing()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  # Reuse the production core and compile only the three IO translation units
  # again for the private process-kill probe. Their object symbols satisfy the
  # static library references, so normal IO objects are not linked into the probe.

  target_include_directories(syrnck_crash_io PRIVATE $<TARGET_PROPERTY:syrnck_core,INCLUDE_DIRECTORIES>)
  target_compile_options(syrnck_crash_io PRIVATE $<TARGET_PROPERTY:syrnck_core,COMPILE_OPTIONS>)
  target_compile_definitions(syrnck_crash_io PRIVATE SYRNCK_TEST_IO_CHECKPOINTS
    $<TARGET_PROPERTY:syrnck_core,COMPILE_DEFINITIONS>)
  get_target_property(SYRNCK_CORE_LINKS syrnck_core LINK_LIBRARIES)
  target_link_libraries(syrnck_crash_io PRIVATE syrnck_core ${SYRNCK_CORE_LINKS})
  add_executable(syrnck_crash_probe tests/crash_probe.cpp $<TARGET_OBJECTS:syrnck_crash_io>)
  target_include_directories(syrnck_crash_probe PRIVATE src)
  target_link_libraries(syrnck_crash_probe PRIVATE syrnck_core)
  add_executable(syrnck_recovery_reader tests/crash_probe.cpp)
  target_include_directories(syrnck_recovery_reader PRIVATE src)
  target_link_libraries(syrnck_recovery_reader PRIVATE syrnck_core)
  add_test(NAME crash_recovery COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/crash_recovery.py
    $<TARGET_FILE:syrnck_crash_probe> $<TARGET_FILE:syrnck_recovery_reader> $<TARGET_FILE:syrnck_ffi>)
  set_tests_properties(crash_recovery PROPERTIES TIMEOUT 240)
  add_executable(syrnck_maintenance_tests tests/maintenance_tests.cpp $<TARGET_OBJECTS:syrnck_crash_io>)
  target_include_directories(syrnck_maintenance_tests PRIVATE src)
  target_link_libraries(syrnck_maintenance_tests PRIVATE syrnck_core)
  add_test(NAME maintenance_operations COMMAND syrnck_maintenance_tests)
  set_tests_properties(maintenance_operations PROPERTIES TIMEOUT 90)
  add_executable(syrnck_core_tests tests/core_tests.cpp)
  target_link_libraries(syrnck_core_tests PRIVATE syrnck_core)
  add_test(NAME core COMMAND syrnck_core_tests)
  add_executable(syrnck_settings_tests tests/settings_tests.cpp)
  target_include_directories(syrnck_settings_tests PRIVATE src)
  target_link_libraries(syrnck_settings_tests PRIVATE syrnck_core)
  add_test(NAME repository_settings COMMAND syrnck_settings_tests)
  add_executable(syrnck_operation_tests tests/operation_tests.cpp)
  target_include_directories(syrnck_operation_tests PRIVATE src)
  target_link_libraries(syrnck_operation_tests PRIVATE syrnck_core syrnck_git)
  add_test(NAME operation_progress COMMAND syrnck_operation_tests)
  set_tests_properties(operation_progress PROPERTIES TIMEOUT 90)
  add_executable(syrnck_audit_operation_tests tests/audit_operation_tests.cpp)
  target_include_directories(syrnck_audit_operation_tests PRIVATE src)
  target_link_libraries(syrnck_audit_operation_tests PRIVATE syrnck_core)
  add_test(NAME audit_operations COMMAND syrnck_audit_operation_tests)
  set_tests_properties(audit_operations PROPERTIES TIMEOUT 90)
  add_executable(syrnck_export_tests tests/export_tests.cpp)
  target_include_directories(syrnck_export_tests PRIVATE src)
  target_link_libraries(syrnck_export_tests PRIVATE syrnck_core)
  add_test(NAME transactional_export COMMAND syrnck_export_tests)
  set_tests_properties(transactional_export PROPERTIES TIMEOUT 90)
  add_executable(syrnck_cli_control_tests tests/cli_control_tests.cpp src/cli_control.cpp)
  target_include_directories(syrnck_cli_control_tests PRIVATE src)
  target_link_libraries(syrnck_cli_control_tests PRIVATE syrnck_core)
  if(MSVC)
    target_compile_options(syrnck_cli_control_tests PRIVATE /utf-8)
    target_compile_definitions(syrnck_cli_control_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  endif()
  add_test(NAME cli_control COMMAND syrnck_cli_control_tests)
  set_tests_properties(cli_control PROPERTIES TIMEOUT 15)
  add_executable(syrnck_i18n_tests tests/i18n_tests.cpp)
  target_include_directories(syrnck_i18n_tests PRIVATE src)
  target_link_libraries(syrnck_i18n_tests PRIVATE syrnck_core)
  if(MSVC)
    target_compile_options(syrnck_i18n_tests PRIVATE /utf-8)
  endif()
  add_test(NAME i18n COMMAND syrnck_i18n_tests)
  add_executable(syrnck_pack_tests tests/pack_tests.cpp)
  target_include_directories(syrnck_pack_tests PRIVATE src)
  target_link_libraries(syrnck_pack_tests PRIVATE syrnck_core)
  add_test(NAME pack_storage COMMAND syrnck_pack_tests)
  set_tests_properties(pack_storage PROPERTIES TIMEOUT 60)

  target_include_directories(syrnck_catalog_tests PRIVATE src)
  target_link_libraries(syrnck_catalog_tests PRIVATE syrnck_core)
  add_test(NAME pack_catalog COMMAND syrnck_catalog_tests)
  set_tests_properties(pack_catalog PROPERTIES TIMEOUT 90)
  add_executable(syrnck_pack_stream_tests tests/pack_stream_tests.cpp)
  target_include_directories(syrnck_pack_stream_tests PRIVATE src)
  target_link_libraries(syrnck_pack_stream_tests PRIVATE syrnck_core BLAKE3::blake3)
  add_test(NAME streaming_pack COMMAND syrnck_pack_stream_tests)
  set_tests_properties(streaming_pack PROPERTIES TIMEOUT 90)
  add_executable(syrnck_index_tests tests/pack_index_tests.cpp)
  target_include_directories(syrnck_index_tests PRIVATE src)
  target_link_libraries(syrnck_index_tests PRIVATE syrnck_core BLAKE3::blake3)
  add_test(NAME paged_pack_index COMMAND syrnck_index_tests)
  set_tests_properties(paged_pack_index PROPERTIES TIMEOUT 90)
  add_executable(syrnck_gc_tests tests/gc_tests.cpp)
  target_include_directories(syrnck_gc_tests PRIVATE src)
  target_link_libraries(syrnck_gc_tests PRIVATE syrnck_core BLAKE3::blake3)
  add_test(NAME streaming_gc COMMAND syrnck_gc_tests)
  set_tests_properties(streaming_gc PROPERTIES TIMEOUT 90)
  add_executable(syrnck_audit_tests tests/audit_tests.cpp)
  target_include_directories(syrnck_audit_tests PRIVATE src)
  target_link_libraries(syrnck_audit_tests PRIVATE syrnck_core)
  add_test(NAME streaming_audit COMMAND syrnck_audit_tests)
  set_tests_properties(streaming_audit PROPERTIES TIMEOUT 90)
  add_executable(syrnck_cache_tests tests/cache_tests.cpp)
  target_link_libraries(syrnck_cache_tests PRIVATE syrnck_core)
  add_test(NAME verified_cache COMMAND syrnck_cache_tests)

  add_executable(syrnck_ingest_tests tests/ingest_tests.cpp)
  target_link_libraries(syrnck_ingest_tests PRIVATE syrnck_core libzstd_static)
  add_test(NAME ingest_pipeline COMMAND syrnck_ingest_tests)
  set_tests_properties(ingest_pipeline PROPERTIES TIMEOUT 60)
  if(SYRNCK_BUILD_CLI)
    if(WIN32)
      add_executable(syrnck_console_runner tests/console_runner.cpp)
      target_compile_definitions(syrnck_console_runner PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
      set(SYRNCK_CONSOLE_RUNNER $<TARGET_FILE:syrnck_console_runner>)

    add_test(NAME cli_operations COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_operations.py

    set_tests_properties(cli_operations PROPERTIES TIMEOUT 90)
    add_executable(syrnck_network_probe tests/network_probe.cpp)
    target_link_libraries(syrnck_network_probe PRIVATE syrnck_net)
    add_test(NAME network_operations COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/network_operations.py
      $<TARGET_FILE:syrnck> $<TARGET_FILE:syrnck_network_probe> ${SYRNCK_CONSOLE_RUNNER})
    set_tests_properties(network_operations PROPERTIES TIMEOUT 90)

  if(SYRNCK_TEST_GO AND NOT SYRNCK_SANITIZE)
    find_program(SYRNCK_GO_EXECUTABLE NAMES go PATHS "${CMAKE_CURRENT_SOURCE_DIR}/.tools/go/bin")
    if(SYRNCK_GO_EXECUTABLE)
      add_test(NAME go_binding COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_go.py
        --go ${SYRNCK_GO_EXECUTABLE} --library $<TARGET_FILE:syrnck_ffi>)
    endif()
  endif()
  foreach(mode shared static)
    add_executable(syrnck_c_${mode}_test tests/c_api_smoke.c)
    if(mode STREQUAL "shared")
      target_link_libraries(syrnck_c_${mode}_test PRIVATE syrnck_ffi)
    else()
      target_link_libraries(syrnck_c_${mode}_test PRIVATE syrnck_ffi_static)
    endif()
    add_test(NAME c_${mode}_link COMMAND syrnck_c_${mode}_test)
  endforeach()
  if(SYRNCK_SANITIZE)
    # Python is not ASan-linked; native C ABI executables exercise the ABI here.
    # The ctypes round trip runs in the normal platform jobs.
    set_tests_properties(c_abi PROPERTIES DISABLED TRUE)
  endif()
