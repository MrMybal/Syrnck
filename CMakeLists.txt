    math(EXPR last "${shards} - 1")
    foreach(shard RANGE 0 ${last})
      add_test(NAME ${name}_${shard} COMMAND ${target} --shard ${shard}/${shards})
      set_tests_properties(${name}_${shard} PROPERTIES TIMEOUT ${timeout} LABELS case-shard)
    endforeach()
    set_property(GLOBAL APPEND PROPERTY SYRNCK_CASE_SUITES
      --suite "${name}" "$<TARGET_FILE:${target}>" "${shards}" "${expected}")
  endfunction()
